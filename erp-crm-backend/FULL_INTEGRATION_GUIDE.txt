STEP 5: CREATE API CLIENT

Create src/api/client.ts:

  import axios from 'axios';

  const API_URL = process.env.REACT_APP_API_URL || 'http://localhost:4000';

  const client = axios.create({
    baseURL: API_URL,
    headers: { 'Content-Type': 'application/json' },
  });

  // Auto-attach JWT
  client.interceptors.request.use((config) => {
    const token = localStorage.getItem('jwt_token');
    if (token) config.headers.Authorization = `Bearer ${token}`;
    return config;
  });

  // Auto-logout on 401
  client.interceptors.response.use(
    (response) => response,
    (error) => {
      if (error.response?.status === 401) {
        localStorage.removeItem('jwt_token');
        window.location.href = '/login';
      }
      return Promise.reject(error);
    }
  );

  export default client;

STEP 6: CREATE API SERVICES

Create src/api/auth.ts:

  import client from './client';

  export const authAPI = {
    login: async (email: string, password: string) => {
      const { data } = await client.post('/auth/login', { email, password });
      localStorage.setItem('jwt_token', data.token);
      return data;
    },

    logout: () => localStorage.removeItem('jwt_token'),
  };

STEP 7: CREATE LOGIN PAGE

Create src/pages/LoginPage.tsx:

  import { useState } from 'react';
  import { useNavigate } from 'react-router-dom';
  import { authAPI } from '../api/auth';

  export function LoginPage() {
    const [email, setEmail] = useState('sales@erp.local');
    const [password, setPassword] = useState('sales123');
    const [error, setError] = useState('');
    const navigate = useNavigate();

    const handleSubmit = async (e: React.FormEvent) => {
      e.preventDefault();
      try {
        await authAPI.login(email, password);
        navigate('/dashboard');
      } catch (err: any) {
        setError(err.response?.data?.message || 'Login failed');
      }
    };

    return (
      <form onSubmit={handleSubmit}>
        <input 
          type="email" 
          value={email} 
          onChange={(e) => setEmail(e.target.value)} 
          placeholder="Email"
        />
        <input 
          type="password" 
          value={password} 
          onChange={(e) => setPassword(e.target.value)} 
          placeholder="Password"
        />
        <button type="submit">Login</button>
        {error && <p style={{ color: 'red' }}>{error}</p>}
      </form>
    );
  }

  import { useState, useEffect } from 'react';
  import { customerAPI } from '../api/customers';

  export function CustomersPage() {
    const [customers, setCustomers] = useState([]);
    const [loading, setLoading] = useState(true);

      <div>

        {loading ? <p>Loading...</p> : (
          <table>
            <thead>
              <tr>
                <th>Name</th>
                <th>Mobile</th>
                <th>Type</th>
                <th>Status</th>
              </tr>
            </thead>
            <tbody>
              {customers.map((c: any) => (
                <tr key={c.id}>
                  <td>{c.name}</td>
                  <td>{c.mobile}</td>
                  <td>{c.customerType}</td>
                  <td>{c.status}</td>
                </tr>
              ))}
            </tbody>
          </table>
        )}
      </div>
    );
  }

STEP 10: SETUP ROUTING

Create src/App.tsx:

  import { BrowserRouter as Router, Routes, Route, Navigate } from 'react-router-dom';
  import { LoginPage } from './pages/LoginPage';
  import { Dashboard } from './pages/Dashboard';
  import { CustomersPage } from './pages/CustomersPage';

  function ProtectedRoute({ children }: { children: React.ReactNode }) {
    const token = localStorage.getItem('jwt_token');
    return token ? children : <Navigate to="/login" />;
  }

  export function App() {
    return (
      <Router>
        <Routes>
          <Route path="/login" element={<LoginPage />} />
          <Route 
            path="/dashboard" 
            element={<ProtectedRoute><Dashboard /></ProtectedRoute>} 
          />
          <Route 
            path="/customers" 
            element={<ProtectedRoute><CustomersPage /></ProtectedRoute>} 
          />
        </Routes>
      </Router>
    );
  }

RESPONSE FORMAT
════════════════════════════════════════════════════════════════════════════

Success (single item):
  { "data": { /* resource */ } }

Success (list):
  { "data": [...], "page": 1, "limit": 20, "total": 100 }

Validation Error (400):
  {
    "error": "validation_error",
    "details": { "fields": { "fieldName": "error message" } }
  }

Insufficient Stock (400):
  {
    "error": "insufficient_stock",
    "details": {
      "details": [
        {
          "productId": "id",
          "productName": "name",
          "requested": 10,
          "available": 5
        }
      ]
    }
  }

Unauthorized (401):
  { "error": "unauthorized", "message": "Invalid or expired token" }

ERROR HANDLING IN FRONTEND
════════════════════════════════════════════════════════════════════════════

Handle validation errors:

  try {
    await customerAPI.create(formData);
  } catch (error: any) {
    const fieldErrors = error.response?.data?.details?.fields || {};
    Object.entries(fieldErrors).forEach(([field, message]: [string, any]) => {
      setFieldError(field, message);
    });
  }

Handle insufficient stock:

  try {
    await challanAPI.confirm(id);
  } catch (error: any) {
    const shortages = error.response?.data?.details?.details || [];
    shortages.forEach((item: any) => {
      console.log(`${item.productName}: need ${item.requested}, have ${item.available}`);
    });
  }

COMMON ISSUES & FIXES
════════════════════════════════════════════════════════════════════════════

Issue: CORS Error "Access blocked by CORS policy"
Fix:
  1. Verify backend is running on http://localhost:4000
  2. Check frontend is on http://localhost:3000 or :5173
  3. Check REACT_APP_API_URL is set in .env.local
  4. Restart frontend (npm start)

Issue: 401 Unauthorized on every request
Fix:
  1. Make sure you're logged in (check localStorage.getItem('jwt_token'))
  2. Try logging in again
  3. Check token is in Authorization header (DevTools Network)

Issue: 404 Not Found
Fix:
  1. Check endpoint path is correct
  2. Verify backend is running
  3. Check baseURL in client.ts

Issue: Connection Refused
Fix:
  1. Start backend: npm run dev
  2. Check port 4000 is not blocked
  3. Verify http://localhost:4000/health returns OK

Issue: CORS in Production
Fix:
  1. Set FRONTEND_URL in backend .env
  2. Update REACT_APP_API_URL to production backend URL
  3. Use HTTPS in production
